cmake_minimum_required(VERSION 3.24)
project(BrickBreakerStage1 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BB_NATIVE "Build native dependencies and diagnostics" OFF)
option(BB_GRAPHICS "Build SDL3/bgfx graphics diagnostic on Windows" ON)
option(BB_GEOMETRY "Build Clipper2 polygon validation" OFF)
include(CTest)
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
add_test(NAME core_tests COMMAND core_tests)
if(BB_NATIVE)
  include(FetchContent)
  set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
  set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
  set(BOX2D_BENCHMARKS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(box2d GIT_REPOSITORY https://github.com/erincatto/box2d.git GIT_TAG v3.1.1 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(box2d)
  add_executable(physics_bench tests/physics_bench.cpp)
  target_include_directories(physics_bench PRIVATE src)
  target_link_libraries(physics_bench PRIVATE box2d)
  add_test(NAME physics_bench COMMAND physics_bench)
  set_tests_properties(physics_bench PROPERTIES TIMEOUT 180)
  if(WIN32 AND BB_GRAPHICS)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(sdl GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-3.2.28 GIT_SHALLOW TRUE)
    set(BGFX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BGFX_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    set(BGFX_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(bgfx_cmake GIT_REPOSITORY https://github.com/bkaradzic/bgfx.cmake.git GIT_TAG v1.171.9651-619 GIT_SUBMODULES_RECURSE TRUE)
    FetchContent_MakeAvailable(sdl bgfx_cmake)
    add_executable(runtime_probe src/runtime_probe.cpp)
    target_link_libraries(runtime_probe PRIVATE SDL3::SDL3-static bgfx)
    install(TARGETS runtime_probe physics_bench RUNTIME DESTINATION .)
  endif()
endif()

if(BB_GEOMETRY)
  include(FetchContent)
  set(CLIPPER2_UTILS OFF CACHE BOOL "" FORCE)
  set(CLIPPER2_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(CLIPPER2_TESTS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(clipper2 GIT_REPOSITORY https://github.com/AngusJohnson/Clipper2.git GIT_TAG Clipper2_1.5.4 GIT_SHALLOW TRUE SOURCE_SUBDIR CPP)
  FetchContent_MakeAvailable(clipper2)
  add_executable(geometry_tests tests/geometry_tests.cpp)
  target_link_libraries(geometry_tests PRIVATE Clipper2)
  add_test(NAME geometry_tests COMMAND geometry_tests)
endif()
